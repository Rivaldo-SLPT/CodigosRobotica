#include <LiquidCrystal.h>

// =================================
// SEMÁFORO DE COCHES
// =================================
const int rojoCoche = 13;
const int amarilloCoche = 12;
const int verdeCoche = 11;

// =================================
// SEMÁFORO DE PEATONES
// =================================
const int rojoPeaton = 10;
const int verdePeaton = 9;

// =================================
// BOTÓN
// =================================
const int boton = 2;

// =================================
// LCD
// RS, E, D4, D5, D6, D7
// =================================
LiquidCrystal lcd(7, 6, 5, 4, 3, 8);

void setup() {

  pinMode(rojoCoche, OUTPUT);
  pinMode(amarilloCoche, OUTPUT);
  pinMode(verdeCoche, OUTPUT);

  pinMode(rojoPeaton, OUTPUT);
  pinMode(verdePeaton, OUTPUT);

  pinMode(boton, INPUT_PULLUP);

  // Inicializar LCD
  lcd.begin(16, 2);

  // Estado inicial
  digitalWrite(verdeCoche, HIGH);
  digitalWrite(amarilloCoche, LOW);
  digitalWrite(rojoCoche, LOW);

  digitalWrite(rojoPeaton, HIGH);
  digitalWrite(verdePeaton, LOW);

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("PUEDE AVANZAR");

  lcd.setCursor(0, 1);
  lcd.print("COCHES: VERDE");
}

void loop() {

  // =================================
  // ESPERAR BOTÓN
  // =================================

  if (digitalRead(boton) == LOW) {

    delay(200);

    // =================================
    // AVISO
    // =================================

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("ESPERE...");

    lcd.setCursor(0, 1);
    lcd.print("CAMBIANDO");

    // =================================
    // AMARILLO PARA COCHES
    // =================================

    digitalWrite(verdeCoche, LOW);
    digitalWrite(amarilloCoche, HIGH);

    delay(2000);

    // =================================
    // ROJO COCHES
    // VERDE PEATONES
    // =================================

    digitalWrite(amarilloCoche, LOW);
    digitalWrite(rojoCoche, HIGH);

    digitalWrite(rojoPeaton, LOW);
    digitalWrite(verdePeaton, HIGH);

    // LCD
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("CRUCE AHORA");

    lcd.setCursor(0, 1);
    lcd.print("PEATON: VERDE");

    delay(5000);

    // =================================
    // VERDE PEATONAL PARPADEA
    // =================================

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("ATENCION!");

    lcd.setCursor(0, 1);
    lcd.print("FIN DEL CRUCE");

    for (int i = 0; i < 3; i++) {

      digitalWrite(verdePeaton, LOW);
      delay(500);

      digitalWrite(verdePeaton, HIGH);
      delay(500);
    }

    // =================================
    // VOLVER A NORMAL
    // =================================

    digitalWrite(verdePeaton, LOW);
    digitalWrite(rojoPeaton, HIGH);

    digitalWrite(rojoCoche, LOW);
    digitalWrite(verdeCoche, HIGH);

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("PUEDE AVANZAR");

    lcd.setCursor(0, 1);
    lcd.print("COCHES: VERDE");

    delay(1000);
  }
}
